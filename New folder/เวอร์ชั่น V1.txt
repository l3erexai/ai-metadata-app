# (โค้ดเหมือนเดิม)
def get_gemini_response(api_key_input, image_data, prompt):
    # ... (เนื้อหาฟังก์ชันเหมือนเดิม) ...
    if not api_key_input: return None
    response = None
    try:
        genai.configure(api_key=api_key_input); model = genai.GenerativeModel('gemini-1.5-flash')
        response = model.generate_content([prompt, image_data])
        if response.candidates and hasattr(response.candidates[0], 'finish_reason') and response.candidates[0].finish_reason == 'SAFETY': return "API_SAFETY_BLOCK"
        if hasattr(response, 'text'): return response.text
        elif response.parts: return "".join(part.text for part in response.parts if hasattr(part, 'text'))
        elif response.candidates and response.candidates[0].content.parts: return "".join(part.text for part in response.candidates[0].content.parts if hasattr(part, 'text'))
        else: return "API_RESPONSE_ERROR"
    except StopCandidateException: return None
    except Exception as e: print(f"Error in get_gemini_response: {e}"); return None

# --- ฟังก์ชันแยก Title และ Keyword (รับค่าจำนวน) ---
def parse_gemini_metadata_v5(text_output, requested_titles, requested_keywords):
    # ... (เนื้อหาฟังก์ชันเหมือนเดิม) ...
    titles = []; keywords = []
    if not text_output or text_output == "API_SAFETY_BLOCK" or text_output == "API_RESPONSE_ERROR": return titles, keywords
    lines = text_output.strip().splitlines(); keyword_line_index = -1; keyword_string = ""
    for i, line in enumerate(lines):
        line_lower_stripped = line.strip().lower()
        if line_lower_stripped.startswith(("keywords:", "tags:", "keywords/tags:")) or (line_lower_stripped.startswith("**") and ("keywords" in line_lower_stripped or "tags" in line_lower_stripped)):
            keyword_line_index = i; keyword_string = re.sub(r"^\*{0,2}(?:Keywords/Tags|Keywords|Tags):\*{0,2}\s*", "", line.strip(), flags=re.IGNORECASE); break
    if keyword_string: keywords = [kw.strip() for kw in keyword_string.split(',') if kw.strip()][:requested_keywords]
    if keyword_line_index != -1:
        for i in range(keyword_line_index):
            line_stripped = lines[i].strip()
            if not line_stripped or line_stripped.lower().startswith(("here's an analysis", "**title suggestion")): continue
            cleaned_title = re.sub(r"^(?:Title:|\d+\.|\*|-)\s*", "", line_stripped, flags=re.IGNORECASE).strip()
            if cleaned_title and len(titles) < requested_titles: titles.append(cleaned_title)
    elif not keywords and len(lines) > 1:
         last_line = lines[-1].strip()
         if ':' not in last_line and ',' in last_line and len(last_line.split(',')) > 2:
              keywords = [kw.strip() for kw in last_line.split(',') if kw.strip()][:requested_keywords]
              for i in range(len(lines) - 1):
                   line_stripped = lines[i].strip();
                   if not line_stripped or line_stripped.lower().startswith(("here's an analysis", "**title suggestion")): continue
                   cleaned_title = re.sub(r"^(?:Title:|\d+\.|\*|-)\s*", "", line_stripped, flags=re.IGNORECASE).strip()
                   if cleaned_title and len(titles) < requested_titles: titles.append(cleaned_title)
    if not titles:
         title_matches_fallback = re.findall(r"^Title:\s*(.*)", text_output, re.MULTILINE | re.IGNORECASE)
         if title_matches_fallback: titles = [match.strip() for match in title_matches_fallback][:requested_titles]
    return titles, keywords

        # --- แสดงผลลัพธ์รวม (จาก final_df ที่ได้จากโหมดใดโหมดหนึ่ง) ---
        if not final_df.empty: # ตรวจสอบว่า DataFrame ไม่ว่าง
            st.subheader("📊 ผลลัพธ์ Metadata:")
            st.dataframe(final_df, hide_index=True, use_container_width=True)
            csv_data = convert_df_to_csv(final_df)
            # กำหนดชื่อไฟล์ดาวน์โหลดตามโหมด
            csv_filename = "gemini_metadata_rerun.csv" if processing_mode == "แก้ไขข้อมูลที่ขาดหายจากไฟล์ CSV ที่อัปโหลด" else "gemini_metadata_export_all.csv"
            st.download_button(
               label=f"📥 ดาวน์โหลดผลลัพธ์ ({'แก้ไขแล้ว' if processing_mode == 'แก้ไขข้อมูลที่ขาดหายจากไฟล์ CSV ที่อัปโหลด' else 'ทั้งหมด'}) เป็น CSV",
               data=csv_data,
               file_name=csv_filename,
               mime='text/csv',
            )
        elif not error_files: # ถ้า DataFrame ว่าง และไม่มี Error อื่นๆ
             st.info("ไม่มีผลลัพธ์ให้แสดง")


        if error_files:
             st.warning("ไฟล์ต่อไปนี้เกิดข้อผิดพลาด หรือถูกบล็อก:")
             for err_file in error_files:
                 st.markdown(f"- `{err_file}`")

# --- แสดงข้อความแนะนำเริ่มต้น ---
elif not uploaded_files:
     st.info("กรุณาอัปโหลดรูปภาพ, กำหนดค่า, เลือกโหมด, แล้วกดปุ่มเริ่ม")
